func TestAddAdding(t *testing.T) {
	test := []struct{ a, b, expected int }{
		{2, 3, 5},
		{0, 0, 0},
		{-1, 1, 1},
	}

	for _, tt := range test {
		result := add(tt.a, tt.b)
		if result != tt.expected {
			t.Errorf("add(%d,%d): expected %d but got %d", tt.a, tt.b, tt.expected, result)
		}

	}
}
